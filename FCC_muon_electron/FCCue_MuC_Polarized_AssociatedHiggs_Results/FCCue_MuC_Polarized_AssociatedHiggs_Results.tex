\bandtitle{1. FCC-ue: 5 TeV \(\mu^+\) on 183 GeV \(e^-\)}
\noindent\textbf{Beam configuration:}\quad \(E_{\mu^+}=5\,\mathrm{TeV},\ E_{e^-}=183\,\mathrm{GeV},\ \sqrt{s}\simeq1.91\,\mathrm{TeV}.\)

\noindent\textbf{Polarization:}\quad \(P_{\mu^+}=+80\%,\quad P_{e^-}=0.\)

\noindent\textbf{Process:}
\begin{equation*}
\mu^+e^-\ \longrightarrow\ \mu^+e^-b\bar b H\ \longrightarrow\ \mu^+e^-b\bar b b\bar b.
\end{equation*}
\crosssections{0.101456}{0.081843\pm0.000255}

\vspace{10pt}
\bandtitle{2. Muon collider: 1 TeV \(\mu^+\) on 1 TeV \(\mu^-\)}
\noindent\textbf{Beam configuration:}\quad \(E_{\mu^+}=1\,\mathrm{TeV},\ E_{\mu^-}=1\,\mathrm{TeV},\ \sqrt{s}=2\,\mathrm{TeV}.\)

\noindent\textbf{Polarization:}\quad \(P_{\mu^+}=+80\%,\quad P_{\mu^-}=0.\)

\noindent\textbf{Process:}
\begin{equation*}
\mu^+\mu^-\ \longrightarrow\ \mu^+\mu^-b\bar b H\ \longrightarrow\ \mu^+\mu^-b\bar b b\bar b.
\end{equation*}
\crosssections{0.116070}{0.094913\pm0.000570}

\vspace{11pt}
\bandtitle{Common fiducial selection}
\noindent\begin{minipage}{\linewidth}
	
\begin{itemize}[leftmargin=18pt,topsep=2pt,itemsep=2pt,parsep=0pt]
\item Four bottom quarks with \(\lvert\eta_b\rvert<5\).
\item Two disjoint \(b\bar b\) invariant-mass windows: \(110<m_{b\bar b}^{(1)},m_{b\bar b}^{(2)}<130\,\mathrm{GeV}\).
\item \textbf{No minimum bottom-quark transverse-momentum cut.}
\item Outgoing charged leptons: \(p_T(\ell)>1\,\mathrm{GeV}\) at generator level.
\end{itemize}
\end{minipage}
