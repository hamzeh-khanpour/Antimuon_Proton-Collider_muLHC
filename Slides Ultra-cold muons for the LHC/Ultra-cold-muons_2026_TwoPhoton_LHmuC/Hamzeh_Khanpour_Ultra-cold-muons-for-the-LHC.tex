\begin{frame}{Why a muon-hadron collider for two-photon physics?}

\begin{itemize}
\justifying
\item A muon-hadron collider combines a clean lepton-hadron initial state with a multi-TeV centre-of-mass energy.
\item Replacing the electron beam by a muon beam strongly relaxes the synchrotron-radiation limitation, making much higher lepton-beam energies possible.
\item The $\mu p$ environment remains experimentally cleaner than $pp$ for many exclusive and semi-exclusive final states: negligible pile-up is a central advantage of the lepton-hadron concept.
\item Both beams radiate photons. Their electromagnetic fields therefore create a high-energy $\gamma\gamma$ collider \emph{inside} the $\mu p$ machine.
\end{itemize}

\begin{alertblock}{A high-energy $\gamma\gamma$ collider!}
\small
\justifying
LH$\mu$C combines a high-energy photon source from the muon beam with the electromagnetic photon field of the proton, opening a distinctive high-energy $\gamma\gamma$ programme.
\end{alertblock}

\begin{frame}{$\gamma\gamma$ at $\ell p$: production topologies and experimental handles}

\begin{columns}%[T,onlytextwidth]
	
\begin{column}{0.55\textwidth}

\begin{block}{Photon-emission topologies}

\begin{itemize}

\footnotesize \justifying
					
\item
\textcolor{LHGreen}{\textbf{Elastic}}
\[
p\rightarrow p+\gamma^{*}
\]
The proton remains intact. Its electromagnetic form factors
$F_E(Q^2)$ and $F_M(Q^2)$ determine the elastic photon flux.
					
\item
\textcolor{LHRed}{\textbf{Proton-dissociative}}
\[
p\rightarrow X_p+\gamma^{*},
\qquad M_N>m_p .
\]

The proton breaks up into a hadronic system $X_p$.
The photon flux is then described using proton structure functions.
Allowing larger proton-dissociation masses $M_N$ and photon virtualities
$Q_p^2$ increases the accessible $\gamma\gamma$ luminosity.					
\end{itemize}
				
\end{block}
			
\end{column}
		

\begin{column}{0.55\textwidth}
			
\begin{alertblock}{Experimental handles}

\begin{itemize}

\footnotesize \justifying
					
\item
\textbf{Low pile-up}
preserves clean final states and makes rapidity-gap /
exclusivity requirements particularly effective.
					
\item
\textbf{Reduced hadronic re-scattering}
compared with $pp$:
there is no second hadronic beam and hence no analogous
soft-survival complication on the lepton side.
					
\item
The \textbf{scattered lepton} constrains the photon emission
from the lepton side; radiative corrections are predominantly QED,
including ISR.
					
\item
A \textbf{forward proton tag}, when available, can over-constrain
the event kinematics and is especially valuable for final states
containing missing energy.
				
\end{itemize}
		
\end{alertblock}
	
\end{column}
		
\end{columns}

\section{Equivalent Photon Approximation and $\gamma\gamma$ luminosity}


\begin{frame}{Equivalent Photon Approximation: factorising the $\ell p$ cross section}
	
\begin{block}{EPA master relation}
\small
\[
\boxed{\displaystyle
\sigma_{\mu p}=\int dy_\mu\,dy_p\;\Phi_\mu(y_\mu)\Phi_p(y_p)\,\sigma_{\gamma\gamma}(W)
      =\int dW\;S_{\gamma\gamma}(W)\,\sigma_{\gamma\gamma}(W)}
\]
\end{block}

\begin{columns}%[T]
	
\begin{column}{0.60\textwidth}
	
\begin{itemize}
\small \justifying 
\item $\Phi_\mu$ and $\Phi_p$ are the equivalent-photon fluxes radiated by the two beams.
\item The hard subprocess $\gamma\gamma\to X$ factorises from photon emission, in close analogy with partonic convolutions.
\item $S_{\gamma\gamma}(W)$ packages the two fluxes into the effective two-photon luminosity spectrum.
\end{itemize}

\end{column}

\begin{column}{0.45\textwidth}
	
\begin{alertblock}{Kinematic variables}
\footnotesize 
\[
y_\mu=\frac{E_\gamma^{(\mu)}}{E_\mu},\qquad
 y_p=\frac{E_\gamma^{(p)}}{E_p},
\]
\[
W\equiv W_{\gamma\gamma}.
\]
\end{alertblock}

\end{column}

\end{columns}


\begin{exampleblock}{Two-photon luminosity spectrum} 
\justifying  \footnotesize
\[
S_{\gamma\gamma}(W)
=
\frac{2}{W}
\int_{W^2/s}^{1}
dy_\ell\,
\Phi_\ell(y_\ell)\,
y_p\,\Phi_p(y_p) = 
\frac{2W}{s}
\int_{W^2/s}^{1}
\frac{dy_\ell}{y_\ell}\,
\Phi_\ell(y_\ell)\,
\Phi_p\!\left(
\frac{W^2}{s\,y_\ell}
\right)
\]
with $y_p=\frac{W^2}{s\,y_\ell}$. Here the standard high-energy EPA relation 
$W^2 \simeq y_\ell y_p s$ is used. 	 
\end{exampleblock}
 
\helplogo 
 
\end{frame}


%==================================================


\begin{frame}{Equivalent photon fluxes from the leptons ($e$ or $\mu^+$) and proton} 
	
\begin{columns}[T]
\begin{column}{0.62\textwidth}
\begin{block}{Generic EPA flux}
\scriptsize
\[
\resizebox{0.96\linewidth}{!}{$\displaystyle
\Phi_\gamma(y)=\frac{\alpha}{\pi y}\int\frac{dQ^2}{Q^2}
\left[(1-y)\left(1-\frac{Q^2_{\min}}{Q^2}\right)F_E(Q^2)
+\frac{y^2}{2}F_M(Q^2)\right]$}
\]
\[
Q^2_{\min}=\frac{M^2y^2}{1-y}.
\]
\end{block}
\begin{itemize}
\small \justifying 
\item Point-like lepton: $F_E=F_M=1$ and $M=m_\mu$.
\item The muon mass raises $Q^2_{\min}$ relative to an electron at the same photon energy fraction $y$.
\end{itemize}
\end{column}

\begin{column}{0.43\textwidth}
	
\begin{alertblock}{Elastic proton emission}
\scriptsize
\[
F_M=G_M^2,
\]
\[
F_E=\frac{4m_p^2G_E^2+Q^2G_M^2}{4m_p^2+Q^2}.
\]
The dipole form factors suppress large $Q_p^2$.
\end{alertblock}

\end{column}
\end{columns}

\begin{alertblock}{Proton-dissociative emission}
\scriptsize	
\[
F_E(Q^2)
=
\int \frac{dx}{x}F_2(x,Q^2),
\qquad
F_M(Q^2)
=
\int \frac{dx}{x^3}F_2(x,Q^2).
\]
	
\footnotesize \justifying 
For \(p\to X_p+\gamma^*\), proton structure functions replace the elastic
form factors; the flux depends on the photon virtuality \(Q_p^2\) and the
dissociative mass \(M_N\):    {\color{blue}$M_N^2 = m_p^2 + Q^2\left(\frac{1}{x}-1\right)$}. 
\end{alertblock}

\helplogo

\end{frame}

\begin{frame}{Beyond the simplest EPA: photon virtuality and exact $W_{\gamma\gamma}$ kinematics}

\vspace{-0.2cm}

\begin{columns}%[T]
	
\begin{column}{0.55\textwidth}
			
\begin{block}{High-energy approximation}
\small
\[
W^2\simeq y_\ell y_p s.
\]
\end{block}

\vspace{-0.3cm}

\begin{alertblock}{Exact elastic and inelastic kinematics}
\small
\textbf{\textcolor{LHGreen}{Elastic:}}
\[
W^2
=
-Q_\ell^2
+
y_\ell y_p\,s
\]
in which depends on the virtuality
$Q_\ell^2$ of the photon emitted by the lepton.
\textbf{\textcolor{LHRed}{Proton-dissociative:}}
\[
\begin{aligned}
W^2 ={}&
-Q_\ell^2-Q_p^2+y_\ell y_p\,s
\\[0.15em]
&+
\frac{Q_\ell^2}{s}
\left(Q_p^2+M_N^2-m_p^2\right)
\end{aligned}
\]
\footnotesize \justifying 
in which depend on the proton-side
photon virtuality $Q_p^2$ and on the invariant mass $M_N$ of the dissociative hadronic system.
	
\end{alertblock}

\end{column}

\begin{column}{0.46\textwidth}
\begin{itemize}
\scriptsize \justifying 
\item The approximation is excellent when photon virtualities are negligible relative to $W^2$.
\item For proton dissociation, the exact expression also depends on $Q_p^2$ and $M_N$.
\item Elastic EPA agrees at percent level with matrix-element generators once the 
proper $W$ definition is used; inelastic modelling is more sensitive to the dissociative phase space.
\end{itemize}

\begin{block}{Precision lesson}
\scriptsize
At sizeable $Q^2$, correct $W_{\gamma\gamma}$ reconstruction is part of the physics model, not a cosmetic refinement.
\end{block}

\end{column}
\end{columns}

\end{frame}

\[
\mathcal{L}_{\rm eff}

+

\[
\mathcal{L}_{\rm aQGC}
=
\sum_j \frac{f_{S,j}}{\Lambda^4}\mathcal{O}_{S,j}

+
\sum_j \frac{f_{T,j}}{\Lambda^4}\mathcal{O}_{T,j}.
\]

\(f_{M2}/\Lambda^4\) is used to illustrate the sensitivity of
$\gamma\gamma\to W^+W^-$ to anomalous quartic interactions.

high-\(M_{WW}\) and high-\(p_T\) tails.

\begin{frame}{Anomalous $\tau$ electromagnetic moments: the $\gamma\tau\tau$ vertex}
	\begin{columns}[T]
		\begin{column}{0.60\textwidth}
			\begin{block}{General electromagnetic vertex}
				\scriptsize
				\[
				\Gamma^\mu(q^2)=\gamma^\mu F_1(q^2)
				+\frac{i\sigma^{\mu\nu}q_\nu}{2m_\tau}F_2(q^2)
				+\sigma^{\mu\nu}q_\nu\gamma_5 F_3(q^2),
				\]
				\[
				F_2(0)=a_\tau,\qquad F_3(0)=-\frac{2m_\tau}{e}\,d_\tau.
				\]
			\end{block}
			\begin{itemize}\scriptsize
				\item $a_\tau$: CP-even anomalous magnetic dipole moment.
				\item $d_\tau$: CP-odd electric dipole moment.
				\item $\gamma\gamma\to\tau^+\tau^-$ probes the vertex directly.
			\end{itemize}
		\end{column}
		\begin{column}{0.36\textwidth}
			\begin{alertblock}{SMEFT interpretation}\tiny
				Gauge-invariant tau dipoles:
				\[
				\resizebox{0.94\linewidth}{!}{$\displaystyle
					\bar L_\tau\sigma^{\mu\nu}\tau_R H
					\left(\frac{C_{\tau B}}{\Lambda^2}B_{\mu\nu}
					+\frac{C_{\tau W}}{\Lambda^2}W_{\mu\nu}\right)+\mathrm{h.c.}$}
				\]
				Real and imaginary coefficient combinations map respectively to magnetic and electric dipole deformations after electroweak symmetry breaking.
			\end{alertblock}
		\end{column}
	\end{columns}
